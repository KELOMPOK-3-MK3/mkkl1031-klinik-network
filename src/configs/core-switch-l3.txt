!
! Core-Switch-L3 — Cisco Catalyst 3560
! Pusat inter-VLAN routing, gerbang setiap VLAN, dan penyalur DHCP ke Server-Layanan.
!
! Alamat gateway mengikuti docs/pengalamatan-vlsm.md:
!   Vlan 10 Pendaftaran  192.168.10.1/26
!   Vlan 20 Poliklinik   192.168.10.65/26
!   Vlan 30 Farmasi      192.168.10.129/27
!   Vlan 40 Laboratorium 192.168.10.161/27
!   Vlan 50 Manajemen    192.168.10.193/28
!   Vlan 99 Tamu         192.168.10.209/28
!   Vlan 100 Server      192.168.10.241/29
!   Gig1/0/1 ke Router-Klinik 172.16.1.2/30
!
! Trafik antar-VLAN diizinkan oleh perangkat ini; pembatasannya dilakukan
! dengan ACL pada bagian bawah berkas.
!
enable
configure terminal
 hostname Core-Switch-L3

 ! --- lapisan tiga wajib aktif pada switch ini ---
 ip routing

 ! ---------------------------------------------------------------- VLAN
 vlan 10
  name PENDAFTARAN
 vlan 20
  name POLIKLINIK
 vlan 30
  name FARMASI
 vlan 40
  name LABORATORIUM
 vlan 50
  name MANAJEMEN
 vlan 99
  name TAMU
 vlan 100
  name SERVER
 vlan 999
  name BUANGAN
 exit

 ! ------------------------------------------------- antarmuka virtual (SVI)
 interface Vlan10
  description Gerbang VLAN Pendaftaran
  ip address 192.168.10.1 255.255.255.192
  no shutdown
 exit
 interface Vlan20
  description Gerbang VLAN Poliklinik
  ip address 192.168.10.65 255.255.255.192
  no shutdown
 exit
 interface Vlan30
  description Gerbang VLAN Farmasi
  ip address 192.168.10.129 255.255.255.224
  no shutdown
 exit
 interface Vlan40
  description Gerbang VLAN Laboratorium
  ip address 192.168.10.161 255.255.255.224
  no shutdown
 exit
 interface Vlan50
  description Gerbang VLAN Manajemen
  ip address 192.168.10.193 255.255.255.240
  no shutdown
 exit
 interface Vlan99
  description Gerbang VLAN Tamu
  ip address 192.168.10.209 255.255.255.240
  no shutdown
 exit
 interface Vlan100
  description Gerbang VLAN Server
  ip address 192.168.10.241 255.255.255.248
  no shutdown
 exit

 ! -------------------------------------------------- tautan ke router tepi
 interface GigabitEthernet1/0/1
  description Tautan ke Router-Klinik
  no switchport
  ip address 172.16.1.2 255.255.255.252
  no shutdown
 exit

 ! ------------------------------------------------------- port trunk ke akses
 interface range GigabitEthernet1/0/2 - 5
  description Trunk ke access switch
  switchport trunk encapsulation dot1q
  switchport trunk native vlan 999
  switchport mode trunk
  switchport trunk allowed vlan 10,20,30,40,50,99,100
  no shutdown
 exit

 ! --------------------------------------------------------- penerusan DHCP
 ! Permintaan DHCP dari tiap VLAN diteruskan ke Server-Layanan.
 ip dhcp relay information trust-all
 interface Vlan10
  ip helper-address 192.168.10.242
 exit
 interface Vlan20
  ip helper-address 192.168.10.242
 exit
 interface Vlan30
  ip helper-address 192.168.10.242
 exit
 interface Vlan40
  ip helper-address 192.168.10.242
 exit
 interface Vlan50
  ip helper-address 192.168.10.242
 exit
 interface Vlan99
  ip helper-address 192.168.10.242
 exit

 ! ---------------------------------------------------------- routing dinamis
 router ospf 1
  router-id 2.2.2.2
  network 192.168.10.0 0.0.0.63 area 0
  network 192.168.10.64 0.0.0.63 area 0
  network 192.168.10.128 0.0.0.31 area 0
  network 192.168.10.160 0.0.0.31 area 0
  network 192.168.10.192 0.0.0.15 area 0
  network 192.168.10.208 0.0.0.15 area 0
  network 192.168.10.240 0.0.0.7 area 0
  network 172.16.1.0 0.0.0.3 area 0
  passive-interface default
  no passive-interface GigabitEthernet1/0/1
 exit

 ! ------------------------------------------------------------- ACL antar VLAN
 ! Aturan dasar:
 !   1. Tamu (VLAN 99) hanya boleh keluar ke internet, tidak ke VLAN internal.
 !   2. Laboratorium dan Farmasi boleh membaca server, tidak sebaliknya.
 !   3. Manajemen boleh menjangkau semua VLAN.
 ! ACL berikut diterapkan pada antarmuka masuk VLAN tamu dan VLAN internal.
 ip access-list extended BATAS_TAMU
  remark tamu hanya boleh keluar lewat router tepi
  permit ip 192.168.10.208 0.0.0.15 host 192.168.10.242
  deny   ip 192.168.10.208 0.0.0.15 192.168.10.0 0.0.0.63
  deny   ip 192.168.10.208 0.0.0.15 192.168.10.64 0.0.0.63
  deny   ip 192.168.10.208 0.0.0.15 192.168.10.128 0.0.0.31
  deny   ip 192.168.10.208 0.0.0.15 192.168.10.160 0.0.0.31
  deny   ip 192.168.10.208 0.0.0.15 192.168.10.192 0.0.0.15
  deny   ip 192.168.10.208 0.0.0.15 192.168.10.240 0.0.0.7
  permit ip any any
 exit
 interface Vlan99
  ip access-group BATAS_TAMU in
 exit

 ! ------------------------------------------------------------ keamanan port
 ! Dua port akses sesuai contoh; sisanya mengikuti pola yang sama.
 interface range GigabitEthernet1/0/2 - 5
  switchport trunk native vlan 999
 exit
 interface range GigabitEthernet1/0/6 - 20
  spanning-tree portfast
  spanning-tree bpduguard enable
 exit

 ! ------------------------------------------------------------------- selesai
end
write memory
