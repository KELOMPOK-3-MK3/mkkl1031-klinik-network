!
! Access Switch 3 — Cisco Catalyst 2960
! Zona tamu (VLAN 99). Tidak ada port manajemen pada switch ini.
!
enable
configure terminal
 hostname SW-Access-3

 vlan 99
  name TAMU
 vlan 999
  name BUANGAN
 exit

 ! ------------------------------------------- trunk ke core switch
 interface GigabitEthernet0/1
  description Trunk ke Core-Switch-L3
  switchport mode trunk
  switchport trunk native vlan 999
  switchport trunk allowed vlan 99
  no shutdown
 exit

 ! ------------------------------------------- port pengguna (tamu)
 interface range FastEthernet0/2 - 20
  description Ruang tunggu tamu
  switchport mode access
  switchport access vlan 99
  spanning-tree portfast
  spanning-tree bpduguard enable
  no shutdown
 exit

 interface range FastEthernet0/21 - 24
  shutdown
 exit

end
write memory
