!
! Access Switch 2 — Cisco Catalyst 2960
! Zona farmasi (VLAN 30), laboratorium (VLAN 40), dan manajemen (VLAN 50).
!
enable
configure terminal
 hostname SW-Access-2

 vlan 30
  name FARMASI
 vlan 40
  name LABORATORIUM
 vlan 50
  name MANAJEMEN
 vlan 999
  name BUANGAN
 exit

 ! ------------------------------------------- trunk ke core switch
 interface GigabitEthernet0/1
  description Trunk ke Core-Switch-L3
  switchport mode trunk
  switchport trunk native vlan 999
  switchport trunk allowed vlan 30,40,50,100
  no shutdown
 exit

 ! ------------------------------------------- port pengguna
 ! Port 2–5: ruang farmasi
 interface range FastEthernet0/2 - 5
  description Ruang farmasi
  switchport mode access
  switchport access vlan 30
  spanning-tree portfast
  spanning-tree bpduguard enable
  no shutdown
 exit

 ! Port 6–10: laboratorium
 interface range FastEthernet0/6 - 10
  description Laboratorium
  switchport mode access
  switchport access vlan 40
  spanning-tree portfast
  spanning-tree bpduguard enable
  no shutdown
 exit

 ! Port 11–14: ruang manajemen
 interface range FastEthernet0/11 - 14
  description Ruang manajemen
  switchport mode access
  switchport access vlan 50
  spanning-tree portfast
  spanning-tree bpduguard enable
  no shutdown
 exit

 interface range FastEthernet0/15 - 24
  description Tidak dipakai — dimatikan
  shutdown
 exit

end
write memory
