!
! Access Switch 1 — Cisco Catalyst 2960
! Zona pendaftaran (VLAN 10) dan poliklinik (VLAN 20).
!
! Port 1 menuju Core-Switch-L3 (trunk), port 2–10 untuk pengguna.
! Salinan ini disalin dari perangkat simulasi setelah diuji.
!
enable
configure terminal
 hostname SW-Access-1

 vlan 10
  name PENDAFTARAN
 vlan 20
  name POLIKLINIK
 vlan 999
  name BUANGAN
 exit

 ! ------------------------------------------- trunk ke core switch
 interface GigabitEthernet0/1
  description Trunk ke Core-Switch-L3
  switchport mode trunk
  switchport trunk native vlan 999
  switchport trunk allowed vlan 10,20,50,100
  no shutdown
 exit

 ! ------------------------------------------- port pengguna
 ! Port 2–6: meja pendaftaran
 interface range FastEthernet0/2 - 6
  description Meja pendaftaran
  switchport mode access
  switchport access vlan 10
  spanning-tree portfast
  spanning-tree bpduguard enable
  no shutdown
 exit

 ! Port 7–10: ruang poliklinik
 interface range FastEthernet0/7 - 10
  description Ruang poliklinik
  switchport mode access
  switchport access vlan 20
  spanning-tree portfast
  spanning-tree bpduguard enable
  no shutdown
 exit

 ! ------------------------------------------- port tidak terpakai dimatikan
 interface range FastEthernet0/11 - 24
  description Tidak dipakai — dimatikan
  shutdown
 exit

end
write memory
